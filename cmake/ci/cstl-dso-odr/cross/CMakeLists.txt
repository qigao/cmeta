cmake_minimum_required(VERSION 3.27)
project(SaltsCstlCrossSdkConsumer LANGUAGES C)
if(NOT WIN32 AND NOT APPLE)
  message(FATAL_ERROR "Win/mac CSTL consumer only")
endif()
if(NOT IS_DIRECTORY "${EXACT_SALTS_PREFIX}")
  message(FATAL_ERROR "EXACT_SALTS_PREFIX must be installed SDK")
endif()
file(REAL_PATH "${EXACT_SALTS_PREFIX}" _prefix)
find_package(Salts CONFIG REQUIRED PATHS "${_prefix}"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
if(NOT TARGET Salts::CSTL)
  message(FATAL_ERROR "missing canonical Salts::CSTL")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
enable_testing()
foreach(role IN ITEMS first second)
  add_library(cstl_cross_${role} SHARED provider.c)
  target_link_libraries(cstl_cross_${role} PRIVATE Salts::CSTL)
  set_target_properties(cstl_cross_${role} PROPERTIES
    WINDOWS_EXPORT_ALL_SYMBOLS OFF)
  if(APPLE)
    set_target_properties(cstl_cross_${role} PROPERTIES
      BUILD_RPATH "${_prefix}/lib")
  endif()
endforeach()
add_executable(cstl_cross_loader loader.c)
add_test(NAME cstl_cross_first COMMAND cstl_cross_loader
  "$<TARGET_FILE:cstl_cross_first>" "$<TARGET_FILE:cstl_cross_second>")
add_test(NAME cstl_cross_second COMMAND cstl_cross_loader
  "$<TARGET_FILE:cstl_cross_second>" "$<TARGET_FILE:cstl_cross_first>")
